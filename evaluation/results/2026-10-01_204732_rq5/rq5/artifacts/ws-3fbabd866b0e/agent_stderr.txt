+"""Small Redis cache adapter using Redis' RESP protocol.
+
+No credentials are hard-coded: connection details are read from REDIS_URL.
+This avoids coupling the project to a particular web framework or Redis
+client package while keeping Redis as the configured data store.
+"""
+
+from __future__ import annotations
+
+import hashlib
+import json
+import os
+import socket
+from dataclasses import dataclass
+from typing import Any
+from urllib.parse import unquote, urlparse
+
+
+class RedisUnavailable(RuntimeError):
+    """Raised internally when Redis cannot service a cache operation."""
+
+
+@dataclass(frozen=True)
+class RedisSettings:
+    host: str
+    port: int
+    password: str | None
+    database: int
+    ttl_seconds: int
+
+    @classmethod
+    def from_environment(cls) -> "RedisSettings":
+        raw_url = os.environ.get("REDIS_URL", "redis://localhost:6379/0")
+        parsed = urlparse(raw_url)
+        if parsed.scheme != "redis" or not parsed.hostname:
+            raise ValueError("REDIS_URL must be a redis:// URL")
+        try:
+            database = int(parsed.path.lstrip("/") or "0")
+            ttl_seconds = int(os.environ.get("REDIS_CACHE_TTL_SECONDS", "300"))
+        except ValueError as exc:
+            raise ValueError("Redis database and cache TTL must be integers") from exc
+        if database < 0 or ttl_seconds <= 0:
+            raise ValueError("Redis database must be non-negative and TTL must be positive")
+        return cls(parsed.hostname, parsed.port or 6379, parsed.password, database, ttl_seconds)
+
+
+class RedisCache:
+    """A Redis-backed JSON cache with graceful cache-miss/outage handling."""
+
+    key_prefix = "untitled-project:echo:"
+
+    def __init__(self, settings: RedisSettings) -> None:
+        self.settings = settings
+
+    @classmethod
+    def from_environment(cls) -> "RedisCache":
+        return cls(RedisSettings.from_environment())
+
+    def get_or_set(self, payload: dict[str, Any]) -> dict[str, Any]:
+        try:
+            encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"))
+        except (TypeError, ValueError) as exc:
+            raise ValueError("payload must be JSON serializable") from exc
+        key = self.key_prefix + hashlib.sha256(encoded.encode()).hexdigest()
+        try:
+            cached = self._command("GET", key)
+            if cached is not None:
+                value = json.loads(cached)
+                if isinstance(value, dict):
+                    return value
+            self._command("SET", key, encoded, "EX", str(self.settings.ttl_seconds))
+        except (OSError, RedisUnavailable, json.JSONDecodeError):
+            # Echo remains available even when an optional cache operation fails.
+            pass
+        return payload
+
+    def _command(self, *parts: str) -> str | None:
+        with socket.create_connection((self.settings.host, self.settings.port), timeout=1.0) as connection:
+            connection.sendall(self._encode_command(*parts))
+            if self.settings.password:
+                connection.sendall(self._encode_command("AUTH", self.settings.password))
+                self._read_response(connection)
+            if self.settings.database:
+                connection.sendall(self._encode_command("SELECT", str(self.settings.database)))
+                self._read_response(connection)
+            return self._read_response(connection)
+
+    @staticmethod
+    def _encode_command(*parts: str) -> bytes:
+        encoded = [part.encode("utf-8") for part in parts]
+        return b"*%d\r\n" % len(encoded) + b"".join(
+            b"$%d\r\n%s\r\n" % (len(part), part) for part in encoded
+        )
+
+    @staticmethod
+    def _read_response(connection: socket.socket) -> str | None:
+        stream = connection.makefile("rb")
+        prefix = stream.read(1)
+        line = stream.readline().rstrip(b"\r\n")
+        if prefix == b"+":
+            return line.decode("utf-8")
+        if prefix == b"-":
+            raise RedisUnavailable(line.decode("utf-8", errors="replace"))
+        if prefix == b"$":
+            length = int(line)
+            if length == -1:
+                return None
+            data = stream.read(length)
+            stream.read(2)
+            return data.decode("utf-8")
+        raise RedisUnavailable("unexpected Redis response")

+"""Small Redis cache adapter using Redis' RESP protocol.
+
+No credentials are hard-coded: connection details are read from REDIS_URL.
+This avoids coupling the project to a particular web framework or Redis
+client package while keeping Redis as the configured data store.
+"""
+
+from __future__ import annotations
+
+import hashlib
+import json
+import os
+import socket
+from dataclasses import dataclass
+from typing import Any
+from urllib.parse import unquote, urlparse
+
+
+class RedisUnavailable(RuntimeError):
+    """Raised internally when Redis cannot service a cache operation."""
+
+
+@dataclass(frozen=True)
+class RedisSettings:
+    host: str
+    port: int
+    password: str | None
+    database: int
+    ttl_seconds: int
+
+    @classmethod
+    def from_environment(cls) -> "RedisSettings":
+        raw_url = os.environ.get("REDIS_URL", "redis://localhost:6379/0")
+        parsed = urlparse(raw_url)
+        if parsed.scheme != "redis" or not parsed.hostname:
+            raise ValueError("REDIS_URL must be a redis:// URL")
+        try:
+            database = int(parsed.path.lstrip("/") or "0")
+            ttl_seconds = int(os.environ.get("REDIS_CACHE_TTL_SECONDS", "300"))
+        except ValueError as exc:
+            raise ValueError("Redis database and cache TTL must be integers") from exc
+        if database < 0 or ttl_seconds <= 0:
+            raise ValueError("Redis database must be non-negative and TTL must be positive")
+        return cls(parsed.hostname, parsed.port or 6379, parsed.password, database, ttl_seconds)
+
+
+class RedisCache:
+    """A Redis-backed JSON cache with graceful cache-miss/outage handling."""
+
+    key_prefix = "untitled-project:echo:"
+
+    def __init__(self, settings: RedisSettings) -> None:
+        self.settings = settings
+
+    @classmethod
+    def from_environment(cls) -> "RedisCache":
+        return cls(RedisSettings.from_environment())
+
+    def get_or_set(self, payload: dict[str, Any]) -> dict[str, Any]:
+        try:
+            encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"))
+        except (TypeError, ValueError) as exc:
+            raise ValueError("payload must be JSON serializable") from exc
+        key = self.key_prefix + hashlib.sha256(encoded.encode()).hexdigest()
+        try:
+            cached = self._command("GET", key)
+            if cached is not None:
+                value = json.loads(cached)
+                if isinstance(value, dict):
+                    return value
+            self._command("SET", key, encoded, "EX", str(self.settings.ttl_seconds))
+        except (OSError, RedisUnavailable, json.JSONDecodeError):
+            # Echo remains available even when an optional cache operation fails.
+            pass
+        return payload
+
+    def _command(self, *parts: str) -> str | None:
+        with socket.create_connection((self.settings.host, self.settings.port), timeout=1.0) as connection:
+            connection.sendall(self._encode_command(*parts))
+            if self.settings.password:
+                connection.sendall(self._encode_command("AUTH", self.settings.password))
+                self._read_response(connection)
+            if self.settings.database:
+                connection.sendall(self._encode_command("SELECT", str(self.settings.database)))
+                self._read_response(connection)
+            return self._read_response(connection)
+
+    @staticmethod
+    def _encode_command(*parts: str) -> bytes:
+        encoded = [part.encode("utf-8") for part in parts]
+        return b"*%d\r\n" % len(encoded) + b"".join(
+            b"$%d\r\n%s\r\n" % (len(part), part) for part in encoded
+        )
+
+    @staticmethod
+    def _read_response(connection: socket.socket) -> str | None:
+        stream = connection.makefile("rb")
+        prefix = stream.read(1)
+        line = stream.readline().rstrip(b"\r\n")

+            return line.decode("utf-8")
+        if prefix == b"-":
+            raise RedisUnavailable(line.decode("utf-8", errors="replace"))

+            length = int(line)
+            if length == -1:
+                return None
+            data = stream.read(length)
+            stream.read(2)
+            return data.decode("utf-8")
+        raise RedisUnavailable("unexpected Redis response")

+"""Small Redis cache adapter using Redis' RESP protocol.
+
+No credentials are hard-coded: connection details are read from REDIS_URL.
+This avoids coupling the project to a particular web framework or Redis
+client package while keeping Redis as the configured data store.
+"""
+
+from __future__ import annotations
+
+import hashlib
+import json
+import os
+import socket
+from dataclasses import dataclass
+from typing import Any
+from urllib.parse import unquote, urlparse
+
+
+class RedisUnavailable(RuntimeError):
+    """Raised internally when Redis cannot service a cache operation."""
+
+
+@dataclass(frozen=True)
+class RedisSettings:
+    host: str
+    port: int
+    password: str | None
+    database: int
+    ttl_seconds: int
+
+    @classmethod
+    def from_environment(cls) -> "RedisSettings":
+        raw_url = os.environ.get("REDIS_URL", "redis://localhost:6379/0")
+        parsed = urlparse(raw_url)
+        if parsed.scheme != "redis" or not parsed.hostname:
+            raise ValueError("REDIS_URL must be a redis:// URL")
+        try:
+            database = int(parsed.path.lstrip("/") or "0")
+            ttl_seconds = int(os.environ.get("REDIS_CACHE_TTL_SECONDS", "300"))
+        except ValueError as exc:
+            raise ValueError("Redis database and cache TTL must be integers") from exc
+        if database < 0 or ttl_seconds <= 0:
+            raise ValueError("Redis database must be non-negative and TTL must be positive")
+        return cls(parsed.hostname, parsed.port or 6379, parsed.password, database, ttl_seconds)
+
+
+class RedisCache:
+    """A Redis-backed JSON cache with graceful cache-miss/outage handling."""
+
+    key_prefix = "untitled-project:echo:"
+
+    def __init__(self, settings: RedisSettings) -> None:
+        self.settings = settings
+
+    @classmethod
+    def from_environment(cls) -> "RedisCache":
+        return cls(RedisSettings.from_environment())
+
+    def get_or_set(self, payload: dict[str, Any]) -> dict[str, Any]:
+        try:
+            encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"))
+        except (TypeError, ValueError) as exc:
+            raise ValueError("payload must be JSON serializable") from exc
+        key = self.key_prefix + hashlib.sha256(encoded.encode()).hexdigest()
+        try:
+            cached = self._command("GET", key)
+            if cached is not None:
+                value = json.loads(cached)
+                if isinstance(value, dict):
+                    return value
+            self._command("SET", key, encoded, "EX", str(self.settings.ttl_seconds))
+        except (OSError, RedisUnavailable, json.JSONDecodeError):
+            # Echo remains available even when an optional cache operation fails.
+            pass
+        return payload
+
+    def _command(self, *parts: str) -> str | None:
+        with socket.create_connection((self.settings.host, self.settings.port), timeout=1.0) as connection:
+            connection.sendall(self._encode_command(*parts))
+            if self.settings.password:
+                connection.sendall(self._encode_command("AUTH", self.settings.password))
+                self._read_response(connection)
+            if self.settings.database:
+                connection.sendall(self._encode_command("SELECT", str(self.settings.database)))
+                self._read_response(connection)
+            return self._read_response(connection)
+
+    @staticmethod
+    def _encode_command(*parts: str) -> bytes:
+        encoded = [part.encode("utf-8") for part in parts]
+        return b"*%d\r\n" % len(encoded) + b"".join(
+            b"$%d\r\n%s\r\n" % (len(part), part) for part in encoded
+        )
+
+    @staticmethod
+    def _read_response(connection: socket.socket) -> str | None:
+        stream = connection.makefile("rb")
+        prefix = stream.read(1)
+        line = stream.readline().rstrip(b"\r\n")

+            return line.decode("utf-8")
+        if prefix == b"-":
+            raise RedisUnavailable(line.decode("utf-8", errors="replace"))
+        if prefix == b"$":
+            length = int(line)
+            if length == -1:
+                return None
+            data = stream.read(length)
+            stream.read(2)
+            return data.decode("utf-8")
+        raise RedisUnavailable("unexpected Redis response")
