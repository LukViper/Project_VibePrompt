+"""Dependency-free WSGI JSON API for the authentication service."""
+from __future__ import annotations
+
+import json
+from typing import Callable
+
+from .auth import AuthenticationError, UserAlreadyExistsError, ValidationError, login, register
+
+
+def _response(start_response: Callable, status: str, body: dict) -> list[bytes]:
+    data = json.dumps(body).encode("utf-8")
+    start_response(status, [("Content-Type", "application/json"), ("Content-Length", str(len(data)))])
+    return [data]
+
+
+def _request_json(environ: dict) -> dict:
+    try:
+        content_length = int(environ.get("CONTENT_LENGTH") or 0)
+    except ValueError as error:
+        raise ValidationError("invalid Content-Length") from error
+    if content_length > 16_384:
+        raise ValidationError("request body is too large")
+    raw_body = environ["wsgi.input"].read(content_length)
+    try:
+        body = json.loads(raw_body.decode("utf-8"))
+    except (UnicodeDecodeError, json.JSONDecodeError) as error:
+        raise ValidationError("request body must be valid JSON") from error
+    if not isinstance(body, dict):
+        raise ValidationError("request body must be a JSON object")
+    return body
+
+
+def application(environ: dict, start_response: Callable) -> list[bytes]:
+    """Serve POST /register and POST /login; no deletion operation exists."""
+    path, method = environ.get("PATH_INFO"), environ.get("REQUEST_METHOD")
+    if method != "POST" or path not in {"/register", "/login"}:
+        return _response(start_response, "404 Not Found", {"error": "not found"})
+    try:
+        body = _request_json(environ)
+        username, password = body.get("username"), body.get("password")
+        if path == "/register":
+            register(username, password)
+            return _response(start_response, "201 Created", {"username": username})
+        return _response(start_response, "200 OK", {"token": login(username, password)})
+    except (ValidationError, UserAlreadyExistsError) as error:
+        return _response(start_response, "400 Bad Request", {"error": str(error)})
+    except AuthenticationError:
+        return _response(start_response, "401 Unauthorized", {"error": "invalid username or password"})
