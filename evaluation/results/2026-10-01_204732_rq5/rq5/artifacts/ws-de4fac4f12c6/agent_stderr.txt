+from __future__ import annotations

+class LogEntry(TypedDict):
+    """A parsed log line."""
+
+    timestamp: str

+    message: str
+
+
+_LOG_LINE_PATTERN = re.compile(

+)
+
+
+def parse_log(line: str) -> LogEntry:
+    """Parse one log line into timestamp, severity, and message fields.
+
+    Args:
+        line: A non-empty string in the form
+            ``<timestamp> <severity> <message>``.  The message may be empty.
+
+    Raises:
+        TypeError: If ``line`` is not a string.
+        ValueError: If the line does not include a timestamp and uppercase
+            severity separated by whitespace.
+    """
+    if not isinstance(line, str):
+        raise TypeError("line must be a string")
+
+    match = _LOG_LINE_PATTERN.fullmatch(line.strip())
+    if match is None:
+        raise ValueError(
+            "line must use the format '<timestamp> <SEVERITY> <message>'"
+        )
+
+    return {
+        "timestamp": match.group("timestamp"),
+        "severity": match.group("severity"),
+        "message": match.group("message") or "",
+    }

 # log_parser
+
+A small, local Python utility for parsing plain-text log lines. It does not
+call external logging APIs.
+
+## Supported format
+
+```
+<timestamp> <SEVERITY> <message>

+"""Utilities for parsing the project's plain-text log format.
+
+The supported format is ``<timestamp> <severity> <message>``.  Parsing is
+entirely local; this module never sends log data to an external service.
+"""
+
+from __future__ import annotations
+
+import re
+from typing import TypedDict
+
+
+class LogEntry(TypedDict):
+    """A parsed log line."""
+
+    timestamp: str
+    severity: str
+    message: str
+
+
+_LOG_LINE_PATTERN = re.compile(
+    r"^(?P<timestamp>\S+)\s+(?P<severity>[A-Z]+)(?:\s+(?P<message>.*))?$"
+)
+
+
+def parse_log(line: str) -> LogEntry:
+    """Parse one log line into timestamp, severity, and message fields.
+
+    Args:
+        line: A non-empty string in the form
+            ``<timestamp> <severity> <message>``.  The message may be empty.
+
+    Raises:
+        TypeError: If ``line`` is not a string.
+        ValueError: If the line does not include a timestamp and uppercase
+            severity separated by whitespace.
+    """
+    if not isinstance(line, str):
+        raise TypeError("line must be a string")
+
+    match = _LOG_LINE_PATTERN.fullmatch(line.strip())
+    if match is None:
+        raise ValueError(
+            "line must use the format '<timestamp> <SEVERITY> <message>'"
+        )
+
+    return {
+        "timestamp": match.group("timestamp"),
+        "severity": match.group("severity"),
+        "message": match.group("message") or "",
+    }
