+"""Minimal Redis cache client using Redis' RESP protocol and the Python stdlib."""
+
+from __future__ import annotations
+
+import hashlib
+import json
+import os
+import socket
+from dataclasses import dataclass
+from typing import Any, Mapping, Protocol
+
+
+class Cache(Protocol):
+    def get(self, payload: Mapping[str, Any]) -> dict[str, Any] | None: ...
+    def set(self, payload: Mapping[str, Any]) -> None: ...
+
+
+@dataclass(frozen=True)
+class RedisSettings:
+    host: str
+    port: int
+    database: int
+    password: str | None
+    timeout_seconds: float
+    ttl_seconds: int
+
+    @classmethod
+    def from_environment(cls) -> "RedisSettings":
+        host = os.getenv("REDIS_HOST", "127.0.0.1").strip()
+        if not host:
+            raise ValueError("REDIS_HOST must not be empty")
+        try:
+            port = int(os.getenv("REDIS_PORT", "6379"))
+            database = int(os.getenv("REDIS_DB", "0"))
+            timeout = float(os.getenv("REDIS_TIMEOUT_SECONDS", "2"))
+            ttl = int(os.getenv("CACHE_TTL_SECONDS", "300"))
+        except ValueError as exc:
+            raise ValueError("Redis environment values must be numeric") from exc
+        if not 1 <= port <= 65535 or database < 0 or timeout <= 0 or ttl <= 0:
+            raise ValueError("Redis environment values are out of range")
+        return cls(host, port, database, os.getenv("REDIS_PASSWORD"), timeout, ttl)
+
+
+class RedisError(RuntimeError):
+    """A Redis protocol or connection failure without secret-bearing detail."""
+
+
+class RedisCache:
+    """A JSON cache persisted solely in Redis."""
+
+    prefix = "student-project:echo:"
+
+    def __init__(self, settings: RedisSettings):
+        self.settings = settings
+
+    def get(self, payload: Mapping[str, Any]) -> dict[str, Any] | None:
+        value = self._execute("GET", self._key(payload))
+        return None if value is None else self._decode_payload(value)
+
+    def set(self, payload: Mapping[str, Any]) -> None:
+        self._execute("SET", self._key(payload), self._encode_payload(payload), "EX", str(self.settings.ttl_seconds))
+
+    def _key(self, payload: Mapping[str, Any]) -> str:
+        return f"{self.prefix}{hashlib.sha256(self._encode_payload(payload)).hexdigest()}"
+
+    @staticmethod
+    def _encode_payload(payload: Mapping[str, Any]) -> bytes:
+        try:
+            return json.dumps(dict(payload), sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8")
+        except (TypeError, ValueError) as exc:
+            raise ValueError("payload must be JSON serializable") from exc
+
+    @staticmethod
+    def _decode_payload(value: bytes) -> dict[str, Any]:
+        try:
+            decoded = json.loads(value.decode("utf-8"))
+        except (UnicodeDecodeError, json.JSONDecodeError) as exc:
+            raise RedisError("Redis contains an invalid cached payload") from exc
+        if not isinstance(decoded, dict):
+            raise RedisError("Redis contains an invalid cached payload")
+        return decoded
+
+    def _execute(self, *parts: str | bytes) -> bytes | None:
+        try:
+            with socket.create_connection((self.settings.host, self.settings.port), self.settings.timeout_seconds) as connection:
+                connection.settimeout(self.settings.timeout_seconds)
+                stream = connection.makefile("rwb")
+                if self.settings.password:
+                    self._write(stream, "AUTH", self.settings.password)
+                    self._read(stream)
+                if self.settings.database:
+                    self._write(stream, "SELECT", str(self.settings.database))
+                    self._read(stream)
+                self._write(stream, *parts)
+                return self._read(stream)
+        except (OSError, EOFError) as exc:
+            raise RedisError("Unable to connect to Redis") from exc
+
+    @staticmethod
+    def _write(stream: Any, *parts: str | bytes) -> None:
+        encoded = [part if isinstance(part, bytes) else part.encode("utf-8") for part in parts]
+        stream.write(f"*{len(encoded)}\r\n".encode())
+        for part in encoded:
+            stream.write(f"${len(part)}\r\n".encode() + part + b"\r\n")
+        stream.flush()
+
+    @staticmethod
+    def _read(stream: Any) -> bytes | None:
+        marker = stream.read(1)
+        if not marker:
+            raise EOFError("Redis closed the connection")
+        line = stream.readline().rstrip(b"\r\n")
+        if marker == b"+":
+            return line
+        if marker == b"-":
+            raise RedisError("Redis command failed")
+        if marker == b"$":
+            length = int(line)
+            if length == -1:
+                return None
+            value = stream.read(length)
+            if len(value) != length or stream.read(2) != b"\r\n":
+                raise RedisError("Malformed Redis response")
+            return value
+        if marker == b":":
+            return line
+        raise RedisError("Unsupported Redis response")
+
+
+_default_cache: RedisCache | None = None
+
+
+def get_default_cache() -> RedisCache:
+    global _default_cache
+    if _default_cache is None:
+        _default_cache = RedisCache(RedisSettings.from_environment())
+    return _default_cache
