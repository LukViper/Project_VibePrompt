+"""A deliberately small, in-memory authentication store.
+
+The store is process-local: restarting the Python process clears all users.
+"""
+
+from __future__ import annotations
+
+import hashlib
+import hmac
+import secrets
+
+
+# Kept public so an embedding application (and tests) can reset the in-memory
+# store.  Passwords are never stored in plaintext.
+USERS: dict[str, tuple[bytes, bytes]] = {}
+_HASH_ITERATIONS = 600_000
+
+
+def _validate_credentials(username: str, password: str) -> None:
+    if not isinstance(username, str) or not username:
+        raise ValueError("username must be a non-empty string")
+    if not isinstance(password, str) or not password:
+        raise ValueError("password must be a non-empty string")
+
+
+def _hash_password(password: str, salt: bytes) -> bytes:
+    return hashlib.pbkdf2_hmac(
+        "sha256", password.encode("utf-8"), salt, _HASH_ITERATIONS
+    )
+
+
+def register(username: str, password: str) -> None:
+    """Create an in-memory user.
+
+    Raises ``ValueError`` for invalid credentials or an already registered
+    username.
+    """
+    _validate_credentials(username, password)
+    if username in USERS:
+        raise ValueError("username is already registered")
+
+    salt = secrets.token_bytes(16)
+    USERS[username] = (salt, _hash_password(password, salt))
+
+
+def login(username: str, password: str) -> str:
+    """Authenticate a user and return a new opaque, non-empty token.
+
+    Raises ``ValueError`` when the credentials do not authenticate.
+    """
+    _validate_credentials(username, password)
