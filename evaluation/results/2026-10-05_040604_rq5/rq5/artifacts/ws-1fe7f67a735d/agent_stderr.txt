+"""Minimal Redis cache adapter using Redis' RESP protocol over a socket.
+
+Using the standard library keeps the project dependency-free while still
+using Redis as its only persistence/cache backend (REQ-003).
+"""
+
+from __future__ import annotations
+
+import os
+import socket
+from dataclasses import dataclass
+from typing import Protocol
+from urllib.parse import unquote, urlparse
+
+
+class Cache(Protocol):
+    def get(self, key: str) -> str | None: ...
+
+    def set(self, key: str, value: str) -> None: ...
+
+
+@dataclass(frozen=True)
+class RedisSettings:
+    host: str
+    port: int
+    password: str | None = None
+    database: int = 0
+    timeout_seconds: float = 2.0
+
+    @classmethod
+    def from_environment(cls) -> "RedisSettings":
+        """Read Redis configuration without ever logging its credentials."""
+        raw_url = os.getenv("REDIS_URL", "redis://127.0.0.1:6379/0")
+        parsed = urlparse(raw_url)
+        if parsed.scheme != "redis" or not parsed.hostname:
+            raise ValueError("REDIS_URL must be a redis:// URL")
+        try:
+            database = int(parsed.path.lstrip("/") or "0")
+            port = parsed.port or 6379
+        except ValueError as exc:
+            raise ValueError("REDIS_URL has an invalid port or database") from exc
+        if database < 0 or port < 1 or port > 65535:
+            raise ValueError("REDIS_URL has an invalid port or database")
+        return cls(parsed.hostname, port, unquote(parsed.password) if parsed.password else None, database)
+
+
+class RedisCache:
+    def __init__(self, settings: RedisSettings):
+        self.settings = settings
+
+    @classmethod
+    def from_environment(cls) -> "RedisCache":
+        return cls(RedisSettings.from_environment())
+
+    def get(self, key: str) -> str | None:
+        response = self._command("GET", key)
+        return None if response is None else str(response)
+
+    def set(self, key: str, value: str) -> None:
+        response = self._command("SET", key, value)
+        if response != "OK":
+            raise RuntimeError("Redis did not confirm cache write")
+
+    def _command(self, *parts: str) -> str | None:
+        with socket.create_connection((self.settings.host, self.settings.port), self.settings.timeout_seconds) as connection:
+            stream = connection.makefile("rwb")
+            if self.settings.password:
+                self._write(stream, "AUTH", self.settings.password)
+                self._read(stream)
+            if self.settings.database:
+                self._write(stream, "SELECT", str(self.settings.database))
+                self._read(stream)
+            self._write(stream, *parts)
+            return self._read(stream)
+
+    @staticmethod
+    def _write(stream: object, *parts: str) -> None:
+        command = [f"*{len(parts)}\r\n".encode()]
+        for part in parts:
+            value = part.encode("utf-8")
+            command.extend((f"${len(value)}\r\n".encode(), value, b"\r\n"))
+        stream.write(b"".join(command))  # type: ignore[attr-defined]
+        stream.flush()  # type: ignore[attr-defined]
+
+    @staticmethod
+    def _read(stream: object) -> str | None:
+        first = stream.read(1)  # type: ignore[attr-defined]
+        line = stream.readline().rstrip(b"\r\n")  # type: ignore[attr-defined]
+        if first == b"+":
+            return line.decode("utf-8")
+        if first == b"$":
+            length = int(line)
+            if length == -1:
+                return None
+            value = stream.read(length)  # type: ignore[attr-defined]
+            stream.read(2)  # type: ignore[attr-defined]
+            return value.decode("utf-8")
+        if first == b"-":
+            raise RuntimeError("Redis error")
+        raise RuntimeError("invalid Redis response")

+"""Minimal Redis cache adapter using Redis' RESP protocol over a socket.
+
+Using the standard library keeps the project dependency-free while still
+using Redis as its only persistence/cache backend (REQ-003).
+"""
+
+from __future__ import annotations
+
+import os
+import socket
+from dataclasses import dataclass
+from typing import Protocol
+from urllib.parse import unquote, urlparse
+
+
+class Cache(Protocol):
+    def get(self, key: str) -> str | None: ...
+
+    def set(self, key: str, value: str) -> None: ...
+
+
+@dataclass(frozen=True)
+class RedisSettings:
+    host: str
+    port: int
+    password: str | None = None
+    database: int = 0
+    timeout_seconds: float = 2.0
+
+    @classmethod
+    def from_environment(cls) -> "RedisSettings":
+        """Read Redis configuration without ever logging its credentials."""
+        raw_url = os.getenv("REDIS_URL", "redis://127.0.0.1:6379/0")
+        parsed = urlparse(raw_url)
+        if parsed.scheme != "redis" or not parsed.hostname:
+            raise ValueError("REDIS_URL must be a redis:// URL")
+        try:
+            database = int(parsed.path.lstrip("/") or "0")
+            port = parsed.port or 6379
+        except ValueError as exc:
+            raise ValueError("REDIS_URL has an invalid port or database") from exc
+        if database < 0 or port < 1 or port > 65535:
+            raise ValueError("REDIS_URL has an invalid port or database")
+        return cls(parsed.hostname, port, unquote(parsed.password) if parsed.password else None, database)
+
+
+class RedisCache:
+    def __init__(self, settings: RedisSettings):
+        self.settings = settings
+
+    @classmethod
+    def from_environment(cls) -> "RedisCache":
+        return cls(RedisSettings.from_environment())
+
+    def get(self, key: str) -> str | None:
+        response = self._command("GET", key)
+        return None if response is None else str(response)
+
+    def set(self, key: str, value: str) -> None:
+        response = self._command("SET", key, value)
+        if response != "OK":
+            raise RuntimeError("Redis did not confirm cache write")
+
+    def _command(self, *parts: str) -> str | None:
+        with socket.create_connection((self.settings.host, self.settings.port), self.settings.timeout_seconds) as connection:
+            stream = connection.makefile("rwb")
+            if self.settings.password:
+                self._write(stream, "AUTH", self.settings.password)
+                self._read(stream)
+            if self.settings.database:
+                self._write(stream, "SELECT", str(self.settings.database))
+                self._read(stream)
+            self._write(stream, *parts)
+            return self._read(stream)
+
+    @staticmethod
+    def _write(stream: object, *parts: str) -> None:
+        command = [f"*{len(parts)}\r\n".encode()]
+        for part in parts:
+            value = part.encode("utf-8")
+            command.extend((f"${len(value)}\r\n".encode(), value, b"\r\n"))
+        stream.write(b"".join(command))  # type: ignore[attr-defined]
+        stream.flush()  # type: ignore[attr-defined]
+
+    @staticmethod
+    def _read(stream: object) -> str | None:
+        first = stream.read(1)  # type: ignore[attr-defined]
+        line = stream.readline().rstrip(b"\r\n")  # type: ignore[attr-defined]
+        if first == b"+":
+            return line.decode("utf-8")
+        if first == b"$":
+            length = int(line)
+            if length == -1:

+            value = stream.read(length)  # type: ignore[attr-defined]
+            stream.read(2)  # type: ignore[attr-defined]
+            return value.decode("utf-8")
+        if first == b"-":
+            raise RuntimeError("Redis error")
+        raise RuntimeError("invalid Redis response")

+"""Minimal Redis cache adapter using Redis' RESP protocol over a socket.
+
+Using the standard library keeps the project dependency-free while still
+using Redis as its only persistence/cache backend (REQ-003).
+"""
+
+from __future__ import annotations
+
+import os
+import socket
+from dataclasses import dataclass
+from typing import Protocol
+from urllib.parse import unquote, urlparse
+
+
+class Cache(Protocol):
+    def get(self, key: str) -> str | None: ...
+
+    def set(self, key: str, value: str) -> None: ...
+
+
+@dataclass(frozen=True)
+class RedisSettings:
+    host: str
+    port: int
+    password: str | None = None
+    database: int = 0
+    timeout_seconds: float = 2.0
+
+    @classmethod
+    def from_environment(cls) -> "RedisSettings":
+        """Read Redis configuration without ever logging its credentials."""
+        raw_url = os.getenv("REDIS_URL", "redis://127.0.0.1:6379/0")
+        parsed = urlparse(raw_url)
+        if parsed.scheme != "redis" or not parsed.hostname:
+            raise ValueError("REDIS_URL must be a redis:// URL")
+        try:
+            database = int(parsed.path.lstrip("/") or "0")
+            port = parsed.port or 6379
+        except ValueError as exc:
+            raise ValueError("REDIS_URL has an invalid port or database") from exc
+        if database < 0 or port < 1 or port > 65535:
+            raise ValueError("REDIS_URL has an invalid port or database")
+        return cls(parsed.hostname, port, unquote(parsed.password) if parsed.password else None, database)
+
+
+class RedisCache:
+    def __init__(self, settings: RedisSettings):
+        self.settings = settings
+
+    @classmethod
+    def from_environment(cls) -> "RedisCache":
+        return cls(RedisSettings.from_environment())
+
+    def get(self, key: str) -> str | None:
+        response = self._command("GET", key)
+        return None if response is None else str(response)
+
+    def set(self, key: str, value: str) -> None:
+        response = self._command("SET", key, value)
+        if response != "OK":
+            raise RuntimeError("Redis did not confirm cache write")
+
+    def _command(self, *parts: str) -> str | None:
+        with socket.create_connection((self.settings.host, self.settings.port), self.settings.timeout_seconds) as connection:
+            stream = connection.makefile("rwb")
+            if self.settings.password:
+                self._write(stream, "AUTH", self.settings.password)
+                self._read(stream)
+            if self.settings.database:
+                self._write(stream, "SELECT", str(self.settings.database))
+                self._read(stream)
+            self._write(stream, *parts)
+            return self._read(stream)
+
+    @staticmethod
+    def _write(stream: object, *parts: str) -> None:
+        command = [f"*{len(parts)}\r\n".encode()]
+        for part in parts:
+            value = part.encode("utf-8")
+            command.extend((f"${len(value)}\r\n".encode(), value, b"\r\n"))
+        stream.write(b"".join(command))  # type: ignore[attr-defined]
+        stream.flush()  # type: ignore[attr-defined]
+
+    @staticmethod
+    def _read(stream: object) -> str | None:
+        first = stream.read(1)  # type: ignore[attr-defined]
+        line = stream.readline().rstrip(b"\r\n")  # type: ignore[attr-defined]

+        if first == b"$":
+            length = int(line)
+            if length == -1:
